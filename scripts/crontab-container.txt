# InfoGet 容器内定时任务配置
# ============================================================
# 由 Dockerfile 安装到 /etc/cron.d/infoget
# ============================================================

# 每 6 小时更新一次 RSS Feed（0:00, 6:00, 12:00, 18:00）
0 */6 * * * appuser cd /app && bash scripts/update_feed.sh --cron >> /var/log/infoget-cron.log 2>&1

# 每天凌晨 3:00 完整爬取（访问文章详情页获取更多信息）
0 3 * * * appuser cd /app && INFOGET_MODE=full bash scripts/update_feed.sh --cron >> /var/log/infoget-cron.log 2>&1

# 每周日凌晨 2:00 输出一次完整统计
0 2 * * 0 appuser curl -s http://localhost:8080/api/stats >> /var/log/infoget-cron.log 2>&1
